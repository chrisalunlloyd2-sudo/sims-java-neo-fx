    def _vliw_execution_bundle(self, text, trace):
        """
        VLIW BUNDLING (Step 3): Processes a 'Very Long Instruction Word' packet.
        Allows multiple parallel file and system operations in a single pass.
        Format: [VLIW_BUNDLE] {json_ops} [/VLIW_BUNDLE]
        """
        bundle_match = re.search(r"\[VLIW_BUNDLE\](.*?)\[/VLIW_BUNDLE\]", text, re.DOTALL)
        if not bundle_match:
            return ""

        trace.append("[VLIW] Detected instruction bundle. Executing VLIW packet...")
        try:
            ops_text = bundle_match.group(1).strip()
            results = []
            for line in ops_text.splitlines():
                if not line.strip(): continue
                op = json.loads(line)
                
                kind = op.get("op", "unknown")
                if kind == "write":
                    filename = op.get("path")
                    content = op.get("content")
                    if is_action_allowed("write", filename, trace):
                        res = create_file(os.path.join(BASE_PATH, filename), content)
                        results.append(f"WRITE({filename}): {res}")
                    else:
                        results.append(f"WRITE({filename}): ACCESS DENIED")
                elif kind == "shell":
                    cmd = op.get("cmd")
                    if is_action_allowed("shell", cmd, trace):
                        res = execute_shell(cmd)
                        results.append(f"SHELL({cmd}): {res}")
                    else:
                        results.append(f"SHELL({cmd}): ACCESS DENIED")
            
            trace.append(f"[VLIW] Bundle executed: {len(results)} operations.")
            return "\n\n--- VLIW_EXECUTION_REPORT ---\n" + "\n".join(results) + "\n---------------------------"
        except Exception as e:
            trace.append(f"[VLIW] Bundle failure: {e}")
            return f"\n\n--- VLIW_ERROR ---\n{e}"

    def _recursive_research(self, command, trace):
        """
        RECURSIVE RESEARCH (Step 4): Performs autonomous multi-hop discovery.
        Reads findings, identifies gaps, and searches again.
        """
        trace.append("[RESEARCH] Initiating recursive discovery cycle...")
        # Hop 1: Initial context gathering
        rust_matches, _ = self._fetch_context(command, trace, limit=10)
        
        # Hop 2: Identify missing links
        bridge_prompt = (
            f"Based on these initial findings for '{command}', identify 3 specific missing links "
            f"or deeper technical questions to investigate:\n\n{rust_matches[:5]}"
        )
        questions = self._chat(LOGICIAN_MODEL, [{"role": "user", "content": bridge_prompt}], trace, "RECURSIVE-LINKER")
        
        # Hop 3: Recursive retrieval for the new questions
        trace.append(f"[RESEARCH] Deepening search on links: {questions[:60]}...")
        deep_matches, _ = self._fetch_context(questions, trace, limit=6)
        
        trace.append("[RESEARCH] Recursive cycle complete. Synthesis ready.")
        return deep_matches

    def _speculative_load(self, next_model):
        """
        SPECULATIVE LOADING (Step 2): Warms up the next expert in the K4000
        while the current one is thinking or the user is typing.
        """
        def warm():
            try:
                # Send a no-op to Ollama to pull weights into RAM/Cache
                ollama.generate(model=next_model, prompt="", keep_alive="10m")
            except Exception: pass
        threading.Thread(target=warm, daemon=True).start()

    def _distill_memory(self, trace):
        """
        DISTILLATION & SYNAPTIC PRUNING (Step 5):
        Implements magnitude-based importance thresholding for memory.
        Removes 'weak' (low information density) synapses.
        """
        trace.append("[SYNAPSE] Initiating biological-scale synaptic pruning")
        history = self.vault.retrieve("", limit=50) # Get larger pool for pruning
        if not history:
            return "Neural pathways are clear. Pruning skipped."
        
        # SYNAPTIC FILTER: Identify high-magnitude (high information) memories
        prompt = (
            "Evaluate the following memories. Assign an 'Importance Magnitude' (0.0-1.0) to each. "
            "Identify memories with magnitude < 0.3 (redundant, low-value) for permanent pruning. "
            "Distill the remaining high-magnitude synapses into a core knowledge packet:\n\n"
            f"{history}"
        )
        distilled_data = self._chat(LOGICIAN_MODEL, [{"role": "user", "content": prompt}], trace, "PRUNER")
        
        # PRUNING: Effectively prune context noise by overwriting the distilled core.
        self.vault.store(f"SYNAPTIC_CORE: {distilled_data}", role="training")
        trace.append("[SYNAPSE] Weak connections removed. High-magnitude core stored.")
        return distilled_data

    def _intercept_algorithms(self, planner_text, trace):
        """
        LAYER 1 INTERCEPTION (Step 6): Scans planner notes for algorithmic tokens.
        Updates the isolated Algorithm Vault.
        """
        # Simple heuristic: Look for PascalCase or snake_case words followed by 'algorithm' or 'logic'
        algo_matches = re.findall(r"(\b[A-Za-z0-9_]{4,}\b)\s+(?:algorithm|logic|implementation)", planner_text, re.IGNORECASE)
        for name in set(algo_matches):
            trace.append(f"[INTERCEPT] Layer 1 Algorithm Detected: {name}")
            # Estimate tokens as word count * 1.3
            tokens = len(planner_text.split()) * 1.3
            record_algorithm_use(name.lower(), "", int(tokens))

    def _dispatch_tools(self, text, trace):
        """
        ACTUATOR LAYER (Step 1): Parses model output for action markers.
        Allows the specialist to actually update text files.
        """
        # Look for [GITHUB_COMPARE: url1, url2]
        github_match = re.search(r"\[GITHUB_COMPARE:\s*(.+?),\s*(.+?)\]", text)
        if github_match:
            repo_a, repo_b = github_match.group(1).strip(), github_match.group(2).strip()
            trace.append(f"[ACTUATOR] Comparing GitHub Repos: {repo_a} vs {repo_b}")
            result = compare_github_repos(repo_a, repo_b)
            return f"\n\nGITHUB_REPORT: {result}"

        # Look for [SHELL: command]
        shell_match = re.search(r"\[SHELL:\s*(.+?)\]", text)
        if shell_match:
            cmd = shell_match.group(1).strip()
            if is_action_allowed("shell", cmd, trace):
                trace.append(f"[ACTUATOR] Running shell: {cmd}")
                result = execute_shell(cmd)
                trace.append(f"[ACTUATOR] Result: {result}")
                return f"\n\nSHELL_LOG: {result}"
            else:
                trace.append(f"[ZERO-TRUST] Blocked unauthorized shell: {cmd}")
                return "\n\nSHELL_LOG: ACCESS DENIED (Manifest Mismatch)"

        # Look for [CREATE_FILE: filename] ... [CONTENT_START] ... [CONTENT_END]
        manifest_match = re.search(r"\[CREATE_FILE:\s*(.+?)\]", text)
        if manifest_match:
            filename = manifest_match.group(1).strip()
            if is_action_allowed("write", filename, trace):
                content_match = re.search(r"\[CONTENT_START\](.*?)\[CONTENT_END\]", text, re.DOTALL)
                if content_match:
                    content = content_match.group(1).strip()
                    trace.append(f"[ACTUATOR] Manifesting file: {filename}")
                    result = create_file(os.path.join(BASE_PATH, filename), content)
                    trace.append(f"[ACTUATOR] Result: {result}")
                    return f"\n\nACTUATOR_LOG: {result}"
            else:
                trace.append(f"[ZERO-TRUST] Blocked unauthorized write: {filename}")
                return "\n\nACTUATOR_LOG: ACCESS DENIED (Manifest Mismatch)"
        return ""

    def process(self, cmd):
        response, _ = self.process_with_trace(cmd)
        return response

    def _evict_kv_cache(self, model_name, trace):
        """
        AFFINE KV-CACHE EVICTION (Step 9):
        Immediately signals the model to drop tokens and free VRAM/RAM
        after a task is completed to maintain infinite context availability.
        """
        trace.append(f"[KV-CACHE] Evicting affine tokens for {model_name}...")
        try:
            # Setting keep_alive to 0 triggers an immediate unload/cache clear in Ollama
            ollama.generate(model=model_name, prompt="", keep_alive=0)
            trace.append("[KV-CACHE] VRAM memory reclaimed.")
        except Exception as e:
            trace.append(f"[KV-CACHE] Eviction failed: {e}")

    def process_with_trace(self, cmd):
        command = cmd.strip()
        trace = []

        if not command:
            return "No command received.", "[AEGIS] Waiting for input."

        if command.startswith("/distill") or "distill yourself" in command.lower():
            return self._distill_memory(trace), "\n".join(trace)

        if command.startswith("/dimon") or "system health" in command.lower():
            return json.dumps(get_dimon_report(), indent=2), "\n".join(trace)

        if command.startswith("/algos") or "algorithm stats" in command.lower():
            return json.dumps(get_algo_stats(), indent=2), "\n".join(trace)

        if command.startswith("/index"):
            return self._run_index(trace)

        if command.startswith("/status"):
            return self._system_status(trace)

        if command.startswith("/vector-status"):
            return self._vector_status(trace)

        if command.startswith("/tasks") or command.startswith("/jobs"):
            return self._tasks_status(trace)

        if command.startswith("/watch-status"):
            return self._watch_status(trace)

        if command.startswith("/watch-on"):
            return self._set_watch(True, trace)

        if command.startswith("/watch-off"):
            return self._set_watch(False, trace)

        if command.startswith("/optimize"):
            return self._optimize_system(trace)

        if self._is_optimize_request(command):
            return self._optimize_system(trace)

        if command.lower().startswith("teach ") or command.lower().startswith("remember "):
            return self._store_training_note(command, trace)

        if self._is_theme_request(command):
            return self._apply_theme_request(command, trace)

        if self._is_training_request(command):
            return self._train_topic(command, trace)

        if command.startswith("/scan-dropbox") or self._is_dropbox_scan_request(command):
            return self._scan_dropbox(trace)

        if self._is_compound_background_request(command):
            return self._queue_maintenance_job(command, trace)

        if self._is_background_research_request(command):
            return self._queue_background_research(command, trace)

        if command.startswith("/research") or self._is_research_request(command):
            return self._run_research(command, trace)

        if self._is_capability_question(command):
            return self._capability_overview(trace)

        if self._is_vector_status_question(command):
            return self._vector_status(trace)

        if self._is_mesh_status_question(command):
            return self._mesh_status(trace)

        if self._is_file_listing_question(command):
            return self._file_listing_overview(trace)

        route = self.mesh.classify(command)
        if route.cacheable:
            cached = self.mesh.get_cached_response(command)
            if cached:
                trace.append("[CACHE] Served response from mesh response cache")
                return cached, "\n".join(trace)

        self._ensure_workspace_index(trace)
        rust_matches, local_history = self._fetch_context(command, trace, route.rag_limit)
        expert_model = route.specialist_model or get_expert_model(command)
        
        # Step 4: Recursive Context for complex research/builder requests
        if route.pattern in {"builder", "research"}:
            recursive_context = self._recursive_research(command, trace)
            rust_matches.extend(recursive_context)

        # SPECULATIVE LOADING: Anticipate next model if possible (Logic/Review is usually next)
        self._speculative_load(LOGICIAN_MODEL if expert_model != LOGICIAN_MODEL else CODER_MODEL)

        kqml_input = wrap_kqml("achieve", command, sender="USER", receiver="PLANNER")
        context_blob = self._build_context(command, local_history, rust_matches)
        
        # DIMON Heartbeat: Inject telemetry into context for self-awareness
        context_blob += f"\n\n--- DIMON TELEMETRY ---\n{json.dumps(get_dimon_report())}"
        
        # ALGO PREFETCH: Inject top-weighted algorithms from the isolated vault
        pref_algos = get_algorithm_prefetch()
        if pref_algos:
            context_blob += "\n\n--- ALGORITHM PREFETCH (WEIGHTED) ---\n"
            for name, content in pref_algos:
                context_blob += f"- {name.upper()}: {content or 'Reference only'}\n"

        # KQML BUS: Append message history to the context blob for agent awareness
        context_blob += f"\n\n--- KQML BUS HISTORY ---\n{get_kqml_history()}"

        # SPECULATIVE DECODING (Step 9): Use the 900mb model to draft the initial logic
        trace.append("[SPECULATIVE] Running RAG-Augmented draft...")
        draft = run_speculative_draft(command, context_blob)
        if draft:
            trace.append(f"[SPECULATIVE] Draft synthesized ({len(draft)} chars). Verifying with {expert_model}...")
            context_blob += f"\n\n--- SPECULATIVE DRAFT ---\n{draft}"

        capability_snapshot = self._capability_snapshot()
        planner_output = self._planner_pass(
            command,
            expert_model,
            local_history,
            rust_matches,
            trace,
            route,
        )
        
        # LAYER 1 INTERCEPTION
        self._intercept_algorithms(planner_output, trace)

        trace.append(f"[SPECIALIST] Handing off to {expert_model}")
        start_time = time.time()
        specialist_output = self._chat(
            expert_model,
            [
                {
                    "role": "system",
                    "content": (
                        "You are the specialist agent inside AEGIS. Solve the request using the "
                        "planner notes and the retrieved context. You have a sharp, witty personality. "
                        "ACTUATOR ACCESS (SINGLE):\n"
                        "- [CREATE_FILE: filename.txt] [CONTENT_START] content [CONTENT_END]\n"
                        "- [SHELL: dir]\n"
                        "- [GITHUB_COMPARE: url1, url2]\n\n"
                        "GITHUB SCAN: To compare or analyze repositories, use the [GITHUB_COMPARE] tool. "
                        "For deep web scanning, describe the target URLs clearly.\n\n"
                        "VLIW BUNDLING (PARALLEL): To execute multiple actions at once, use:\n"
                        "[VLIW_BUNDLE]\n"
                        '{"op": "write", "path": "file1.txt", "content": "..."}\n'
                        '{"op": "shell", "cmd": "..."}\n'
                        "[/VLIW_BUNDLE]\n\n"
                        "RECURSIVE THINKING: If the context seems incomplete, mention what is missing "
                        "and propose a deeper search. Be practical and mention concrete actions taken."
                    ),
                },
                {
                    "role": "user",
                    "content": (
                        f"PLANNER NOTES:\n{planner_output}\n\n"
                        f"CURRENT CAPABILITIES:\n{capability_snapshot}\n\n"
                        f"RETRIEVED CONTEXT:\n{context_blob}\n\n"
                        f"REQUEST:\n{command}"
                    ),
                },
            ],
            trace,
            "SPECIALIST",
        )
        update_dimon(expert_model, time.time() - start_time)

        # ACTUATOR & VLIW EXECUTION
        tool_log = self._dispatch_tools(specialist_output, trace)
        vliw_log = self._vliw_execution_bundle(specialist_output, trace)
        specialist_output += f"{tool_log}{vliw_log}"

        # ceLORA SYNTHESIS (Step 9): Log the interaction for weekly weight adaptation
        synth_note = synth_celora_interaction(command, specialist_output)
        trace.append(synth_note)

        # AFFINE EVICTION: Clear cache to reclaim VRAM/RAM if response was large
        if len(specialist_output) > 1500:
            self._evict_kv_cache(expert_model, trace)

        if route.use_guardian:
            trace.append(f"[AEGIS] Final review with {GUARD_MODEL}")
            final_output = self._chat(
                GUARD_MODEL,
                [
                    {
                        "role": "system",
                        "content": (
                            "You are AEGIS, the guardian agent. Merge the planner and specialist outputs, "
                            "fix weak spots, catch contradictions, and answer in a clean tactical briefing. "
                            "Prefer direct, thorough, execution-oriented responses over tutorials. "
                            "Do not assess the user's motives or ask for a moral framework. Do not ask the user questions. "
                            "Keep safety boundaries internally, but present the answer directly and professionally. "
                            "If the specialist overstates a capability, correct it and stay faithful to the actual system state."
                        ),
                    },
                    {
                        "role": "user",
                        "content": (
                            f"USER REQUEST:\n{command}\n\n"
                            f"CURRENT CAPABILITIES:\n{capability_snapshot}\n\n"
                            f"PLANNER:\n{planner_output}\n\n"
                            f"SPECIALIST:\n{specialist_output}\n\n"
                            f"RETRIEVED CONTEXT:\n{context_blob}\n\n"
                            f"KQML ENVELOPE:\n{kqml_input}"
                        ),
                    },
                ],
                trace,
                "AEGIS",
            )
        else:
            trace.append("[AEGIS] Guardian pass skipped by mesh optimizer")
            final_output = specialist_output

        final_output = self._ensure_non_empty_output(final_output, specialist_output, planner_output, trace)
        final_output = self._reduce_user_interrogation(final_output, trace)
        self.vault.store(command, role="user")
        self.vault.store(final_output, role="assistant")
        trace.append("[MEMORY] Stored latest exchange in local memory")

        if route.cacheable:
            self.mesh.cache_response(command, final_output)

        return final_output, "\n".join(trace)

    def _run_index(self, trace):
        trace.append("[RUST-RAG] Reindexing workspace")
        result = self.rag.index_workspace()
        self.workspace_indexed = True
        summary = (
            "Rust retrieval index refreshed.\n"
            f"Files indexed: {result.get('indexed_files', 0)}\n"
            f"Chunks available: {result.get('chunks', 0)}"
        )
        return summary, "\n".join(trace)

    def _system_status(self, trace):
        memory_status = self.vault.status()
        status_lines = [
            f"Conversation vector store: {memory_status.get('backend')} ({memory_status.get('count', 0)} entries)",
            f"Conversation vector path: {memory_status.get('path')}",
        ]
        if not memory_status.get("persistent", False):
            status_lines.append("Conversation vector note: persistent store is already owned by the live UI process, so this helper session is using a safe fallback.")
        try:
            rag_stats = self.rag.stats()
            status_lines.append(
                f"Rust RAG chunks: {rag_stats.get('chunks', 0)} from {rag_stats.get('sources', 0)} sources"
            )
            if "embedding_backend" in rag_stats:
                status_lines.append(
                    f"Rust vector backend: {rag_stats.get('embedding_backend')} ({rag_stats.get('vector_size', 'unknown')} dims)"
                )
            if "cache_entries" in rag_stats:
                status_lines.append(
                    f"Rust RAG cache: {rag_stats.get('cache_entries', 0)} entries, {rag_stats.get('cache_hits', 0)} hits"
                )
        except Exception as exc:
            trace.append(f"[RUST-RAG] Status unavailable: {exc}")
            status_lines.append("Rust RAG: unavailable")

        provider = self._active_cloud_provider()
        status_lines.append(f"Cloud fallback: {provider or 'disabled'}")
        status_lines.append(f"Direct execution mode: {'enabled' if DIRECT_EXECUTION_MODE else 'disabled'}")
        status_lines.append(f"Tutorial mode: {'enabled' if TUTORIAL_MODE else 'disabled'}")
        status_lines.append(f"Desktop drop folder: {self.dropbox_dir}")
        status_lines.append(f"Drop-folder files detected: {len(list_dropbox_files())}")
        watcher = self.tasks.watcher_status
        status_lines.append(
            f"Drop-folder watch: {'enabled' if watcher.get('enabled') else 'disabled'} every {watcher.get('interval_secs')}s"
        )
        status_lines.append(f"Watch note: {watcher.get('last_note', 'n/a')}")
        running = sum(1 for task in self.tasks.list_tasks(limit=20) if task.status == "running")
        queued = sum(1 for task in self.tasks.list_tasks(limit=20) if task.status == "queued")
        status_lines.append(f"Background tasks: {running} running, {queued} queued")
        return "\n".join(status_lines), "\n".join(trace or ["[AEGIS] Status collected"])

    def _set_watch(self, enabled, trace):
        if enabled:
            self.tasks.start_watcher(self._watch_dropbox_once)
            message = "Drop-folder watch is now enabled."
        else:
            self.tasks.stop_watcher()
            message = "Drop-folder watch is now disabled."
        trace.append(f"[WATCH] {message}")
        return message, "\n".join(trace)

    def _optimize_system(self, trace):
        trace.append("[AEGIS] Running self-optimize routine")
        self.workspace_indexed = False
        index_result = self.rag.index_workspace()
        optimize_result = self.rag.optimize()
        response = "\n".join(

    def _perform_research(self, command, trace, force_budget=None):
        research_request = self._normalize_research_request(command)
        budget = force_budget or parse_research_budget(command)
        deep_mode = self._is_deep_research(command)
        include_dropbox = self._should_use_dropbox(command)
        trace.append("[RESEARCH] Defining scope from the plain-language request")
        trace.append(f"[RESEARCH] Budget set to {budget.label}")

        bundle = collect_research_materials(
            research_request,
            include_dropbox=include_dropbox,
            budget=budget,
            auto_discover=True,
        )
        if bundle.cache_dir:
            trace.append(f"[RESEARCH] Cached source packet at {bundle.cache_dir}")
        trace.append(
            f"[RESEARCH] Pulled {len(bundle.urls)} direct URLs and {len(bundle.discovered_urls)} discovered URLs"
        )
        if bundle.errors:
            trace.append(f"[RESEARCH] Source fetch notes: {len(bundle.errors)}")

        if include_dropbox and bundle.dropbox_files:
            changed_documents = [document for document in bundle.documents if document.source.startswith("dropbox::")]
            if changed_documents:
                self._store_documents(changed_documents, "dropbox", trace, label="drop-folder files", optimize_after=False)

        non_dropbox_documents = [
            document for document in bundle.documents if not document.source.startswith("dropbox::")
        ]
        if non_dropbox_documents:
            self._store_documents(non_dropbox_documents, "research-sources", trace, label="research sources", optimize_after=False)

        if bundle.documents:
            try:
                self.rag.optimize()
                trace.append("[RESEARCH] Warmed Rust retrieval after source ingestion")
            except Exception as exc:
                trace.append(f"[RESEARCH] Optimize after source ingestion failed: {exc}")

        self._ensure_workspace_index(trace)
        rust_matches, local_history = self._fetch_context(
            research_request,
            trace,
            max(RAG_QUERY_LIMIT + 2, 8),
        )
        scope_output = self._build_research_scope_outline(research_request, bundle, rust_matches, budget)

        report_body = self._assemble_research_report(
            research_request,
            scope_output,
            bundle,
            rust_matches,
            deep_mode,
        )

        request_slug = self._slugify(research_request[:100])
        report_title = f"Research Packet: {research_request[:120].strip()}"
        source_list = [document.source for document in bundle.documents]
        report_path = write_research_report(request_slug, report_title, report_body, sources=source_list)
        self.vault.store(report_body, role="research")
        self.rag.store(report_body, source=f"research::{request_slug}")
        trace.append(f"[RESEARCH] Saved research packet to {report_path}")

        summary_lines = [
            report_body,
            "",
            f"Research budget: {budget.label}",
            f"Direct URLs pulled: {len(bundle.urls)}",
            f"Discovered URLs pulled: {len(bundle.discovered_urls)}",
            f"Saved research packet: {report_path}",
            f"Source cache: {bundle.cache_dir or RESEARCH_REPORTS_DIR}",
            f"Desktop drop folder: {self.dropbox_dir}",
        ]
        return {
            "response": "\n".join(summary_lines),
            "summary": f"Research packet saved with {len(bundle.documents)} sources.",
            "report_path": str(report_path),
            "budget_label": budget.label,
            "bundle": bundle,
        }

    def _store_documents(self, documents, manifest_name, trace=None, label="documents", optimize_after=False):
        changed_documents, manifest = diff_documents_against_manifest(manifest_name, documents)
        if not changed_documents:
            if trace is not None:
                trace.append(f"[INGEST] No new or changed {label}")
            return []

        for document in changed_documents:
            self.rag.store(document.text, source=document.source)
        save_document_manifest(manifest_name, manifest)
        if trace is not None:
            trace.append(f"[INGEST] Stored {len(changed_documents)} new or changed {label}")

        if optimize_after:
            try:
                self.rag.optimize()
                if trace is not None:
                    trace.append("[INGEST] Warmed Rust retrieval after ingestion")
            except Exception as exc:
                if trace is not None:
                    trace.append(f"[INGEST] Optimize after ingestion failed: {exc}")
        return changed_documents

    def _ingest_dropbox_documents(self, trace=None):
        if trace is not None:
            trace.append(f"[DROPBOX] Scanning {self.dropbox_dir}")
        documents = load_dropbox_documents()
        if not documents:
            return {"scanned": 0, "ingested": 0, "ingested_sources": []}

        changed_documents = self._store_documents(
            documents,
            "dropbox",
            trace=trace,
            label="drop-folder files",
            optimize_after=True,
        )
        return {
            "scanned": len(documents),
            "ingested": len(changed_documents),
            "ingested_sources": [document.source.replace("dropbox::", "", 1) for document in changed_documents],
        }

    def _watch_dropbox_once(self):
        result = self._ingest_dropbox_documents(trace=[])
        return result.get("ingested", 0)

    def _scan_local_roots(self, trace=None, progress_log=None):
        total_scanned = 0
        total_changed = 0
        changed_sources = []

        for root in LOCAL_SCAN_ROOTS:
            prefix = f"localscan-{self._slugify(str(root))}"
            documents = load_documents_from_directory(root, prefix=prefix, max_files=60, char_limit=14000)
            total_scanned += len(documents)
            changed = self._store_documents(
                documents,
                f"localscan-{self._slugify(str(root))}",
                trace=trace,
                label=f"local files from {root}",
                optimize_after=False,
            )
            total_changed += len(changed)
            changed_sources.extend(document.source for document in changed[:10])
            if progress_log is not None:
                progress_log(
                    f"Local root {root} scanned={len(documents)} ingested={len(changed)}"
                )

        if total_changed:
            try:
                self.rag.optimize()
                if trace is not None:
                    trace.append("[LOCALS] Warmed Rust retrieval after local directory scan")
                if progress_log is not None:
                    progress_log("Local directory scan warmed Rust retrieval")
            except Exception as exc:
                if trace is not None:
                    trace.append(f"[LOCALS] Optimize after local scan failed: {exc}")
                if progress_log is not None:
                    progress_log(f"Local directory scan optimize failed: {exc}")

        return {
            "roots": [str(path) for path in LOCAL_SCAN_ROOTS],
            "scanned": total_scanned,
            "ingested": total_changed,
            "changed_sources": changed_sources,
        }

    def _queue_background_research(self, command, trace):
        budget = parse_research_budget(command)
        title = self._normalize_research_request(command)[:120] or "background research"

        def runner(task_id, log):
            passes, interval_secs = self._background_research_schedule(budget.requested_minutes)
            latest = None
            for index in range(1, passes + 1):
                pass_trace = []
                log(task_id, f"Research pass {index}/{passes} started")
                latest = self._perform_research(command, pass_trace, force_budget=budget)
                for line in pass_trace[-12:]:
                    log(task_id, line)
                if index < passes:
                    log(task_id, f"Sleeping {interval_secs} seconds before next pass")
                    time.sleep(interval_secs)
            return {
                "summary": latest["summary"] if latest else "Background research completed.",
                "result_path": latest["report_path"] if latest else "",
            }

        task = self.tasks.submit(
            kind="research",
            title=title,
            command=command,
            runner=runner,
            budget_label=budget.label,
        )
        trace.append(f"[TASKS] Queued background research job {task.id}")
        response = "\n".join(
            [
                f"Background research job started: `{task.id}`",
                f"Budget: `{budget.label}`",
                "Use `/tasks` or `/status` to watch progress.",
            ]
        )
        return response, "\n".join(trace)

    def _queue_maintenance_job(self, command, trace):
        budget = parse_research_budget(command)
        title = "maintenance sweep"

        def runner(task_id, log):
            passes, interval_secs = self._background_research_schedule(budget.requested_minutes)
            latest = None
            last_drop_result = {"ingested": 0}
            last_local_result = {"ingested": 0}
            warnings = []
            lowered = command.lower()
            wants_full_reindex = any(
                marker in lowered
                for marker in ["reindex workspace", "full reindex", "rebuild workspace index"]
            )

            for index in range(1, passes + 1):
                log(task_id, f"Maintenance pass {index}/{passes} started")
                log(task_id, "Clearing mesh caches")
                self.mesh.clear_caches()

                drop_result = self._ingest_dropbox_documents(trace=[])
                last_drop_result = drop_result
                log(
                    task_id,
                    f"Drop-folder scanned={drop_result.get('scanned', 0)} ingested={drop_result.get('ingested', 0)}",
                )

                local_trace = []
                local_result = self._scan_local_roots(
                    trace=local_trace,
                    progress_log=lambda message: log(task_id, message),
                )
                last_local_result = local_result
                for line in local_trace[-10:]:
                    log(task_id, line)
                log(
                    task_id,
                    f"Local roots scanned={local_result.get('scanned', 0)} ingested={local_result.get('ingested', 0)}",
                )

                if wants_full_reindex:
                    try:
                        index_result = self.rag.index_workspace()
                        log(task_id, f"Workspace reindexed with {index_result.get('chunks', 0)} chunks")
                    except Exception as exc:
                        warning = f"Workspace reindex skipped after failure: {exc}"
                        warnings.append(warning)
                        log(task_id, warning)
                else:
                    try:
                        rag_stats = self.rag.stats()
                        if rag_stats.get("chunks", 0) == 0:
                            index_result = self.rag.index_workspace()
                            log(task_id, f"Workspace index was empty, rebuilt with {index_result.get('chunks', 0)} chunks")
                        else:
                            log(task_id, "Workspace index already available, skipped full reindex")
                    except Exception as exc:
                        warning = f"Workspace index check failed: {exc}"
                        warnings.append(warning)
                        log(task_id, warning)

                try:
                    optimize_result = self.rag.optimize()
                    log(task_id, f"Rust optimize complete with {optimize_result.get('cache_entries', 0)} cache entries")
                except Exception as exc:
                    warning = f"Rust optimize failed: {exc}"
                    warnings.append(warning)
                    log(task_id, warning)

                if "research" in command.lower():
                    try:
                        research_trace = []
                        latest = self._perform_research(command, research_trace, force_budget=budget)
                        for line in research_trace[-12:]:
                            log(task_id, line)
                    except Exception as exc:
                        warning = f"Research phase failed: {exc}"
                        warnings.append(warning)
                        log(task_id, warning)

                if index < passes:
                    log(task_id, f"Sleeping {interval_secs} seconds before next maintenance pass")
                    time.sleep(interval_secs)

            summary = (
                f"Maintenance complete after {passes} pass(es). "
                f"Drop-folder ingested {last_drop_result.get('ingested', 0)} files; "
                f"local scan ingested {last_local_result.get('ingested', 0)} files."
            )
            result_path = latest["report_path"] if latest else ""
            if latest:
                summary += f" Research packet saved to {result_path}."
            if warnings:
                summary += " Warnings: " + " | ".join(warnings[:3])
            return {"summary": summary, "result_path": result_path}

        task = self.tasks.submit(
            kind="maintenance",
            title=title,
            command=command,
            runner=runner,
            budget_label=budget.label,
        )
        trace.append(f"[TASKS] Queued maintenance job {task.id}")
        response = "\n".join(
            [
                f"Background maintenance job started: `{task.id}`",
                f"Budget: `{budget.label}`",
                "This job will research, optimize, clean caches, scan the drop folder, and scan the configured local roots.",
                "Use `/tasks` or `/status` to watch progress.",
            ]
        )
        return response, "\n".join(trace)

    def _background_research_schedule(self, minutes):
        if minutes <= 30:
            return 1, 0
        if minutes <= 180:
            passes = 2
        elif minutes <= 720:
            passes = 3
        elif minutes <= 1440:
            passes = 4
        else:
            passes = 6
        interval_secs = max(300, int((minutes * 60) / passes))
        return passes, interval_secs

    def _ensure_workspace_index(self, trace):
        if self.workspace_indexed:
            return
        try:
            stats = self.rag.stats()
            if stats.get("chunks", 0) == 0:
                trace.append("[RUST-RAG] Empty index detected, building workspace index")
                self.rag.index_workspace()
            self.workspace_indexed = True
            trace.append("[RUST-RAG] Workspace index ready")
        except Exception as exc:
            trace.append(f"[RUST-RAG] Index check failed: {exc}")

    def _fetch_context(self, command, trace, rag_limit):
        try:
            rust_matches, local_history = self.mesh.fetch_context(
                command,
                self._load_rag_matches,
                self.vault.retrieve,
                rag_limit,
            )
            trace.append(f"[RUST-RAG] Retrieved {len(rust_matches)} relevant chunks")
            trace.append(f"[MEMORY] Retrieved {len(local_history)} local memory entries")
            return rust_matches, local_history
        except Exception as exc:
            trace.append(f"[MESH] Context fetch failed: {exc}")
            return [], self.vault.retrieve(command)

    def _load_rag_matches(self, command, limit):
        result = self.rag.retrieve(command, limit=limit)
        return result.get("matches", [])

    def _build_context(self, command, local_history, rust_matches):
        sections = []
        prefetched = self.mesh.get_prefetch(command)

        if local_history:
            sections.append("LOCAL MEMORY")
            sections.extend(f"- {item}" for item in local_history[:4])

        if rust_matches:
            sections.append("RUST RETRIEVAL")
            for match in rust_matches:
                source = match.get("source", "unknown")
                excerpt = match.get("excerpt", "")
                sections.append(f"- {source}: {excerpt}")

        if prefetched:
            sections.append("PREFETCHED CONTEXT")
            for match in prefetched[:3]:
                source = match.get("source", "unknown")
                excerpt = match.get("excerpt", "")
                sections.append(f"- {source}: {excerpt}")

        if not sections:
            sections.append("No supporting context available yet.")

        return "\n".join(sections)

    def _chat(self, model, messages, trace, agent_name, num_ctx=None, max_predict=None, temperature=None):
        options = {
            "num_ctx": num_ctx or CONTEXT_WINDOW,
            "temperature": TEMPERATURE if temperature is None else temperature,
            "num_predict": max_predict or MAX_PREDICT,
            "num_thread": NUM_THREADS,
        }
        try:
            response = ollama.chat(
                model=model,
                messages=messages,
                options=options,
            )
            text = response["message"]["content"].strip()
            if text:
                return text

            trace.append(f"[{agent_name}] Empty local model response, retrying once")
            retry_messages = list(messages) + [
                {
                    "role": "user",
                    "content": "Your last answer came back empty. Respond again with a complete helpful answer.",
                }
            ]
            retry = ollama.chat(
                model=model,
                messages=retry_messages,
                options=options,
            )
            return retry["message"]["content"].strip()
        except Exception as exc:
            trace.append(f"[{agent_name}] Local model failed: {exc}")
            cloud_response = self._cloud_completion(messages, trace, agent_name)
            if cloud_response:
                return cloud_response
            raise

    def _ensure_non_empty_output(self, final_output, specialist_output, planner_output, trace):
        if final_output and final_output.strip():
            return final_output.strip()
        if specialist_output and specialist_output.strip():
            trace.append("[AEGIS] Final output was empty, falling back to specialist output")
            return specialist_output.strip()
        if planner_output and planner_output.strip():
            trace.append("[AEGIS] Final and specialist outputs were empty, falling back to planner output")
            return f"AEGIS fallback response:\n{planner_output.strip()}"
        trace.append("[AEGIS] All model stages returned empty output")
        return (
            "I did not get a usable response from the model on that pass. "
            "Please retry the same request."
        )

    def _reduce_user_interrogation(self, text, trace):
        if not text:
            return text

        cleaned = text
        patterns = [
            r"(?im)^\s*would you like[^\n]*\n?",
            r"(?im)^\s*do you want[^\n]*\n?",
            r"(?im)^\s*should i[^\n]*\n?",
            r"(?im)^\s*shall i[^\n]*\n?",
            r"(?im)^\s*can you[^\n]*\n?",
            r"(?im)^\s*could you[^\n]*\n?",
            r"(?im)^\s*if you want, i can[^\n]*\n?",
        ]
        for pattern in patterns:
            cleaned = re.sub(pattern, "", cleaned)

        cleaned = re.sub(r"\n{3,}", "\n\n", cleaned).strip()
        if cleaned != text:
            trace.append("[AEGIS] Removed follow-up questions to keep execution direct")
        if cleaned.endswith("?"):
            cleaned = cleaned[:-1] + "."
        return cleaned

    def _cloud_completion(self, messages, trace, agent_name):
        provider = self._active_cloud_provider()
        if not provider:
            return None

        trace.append(f"[{agent_name}] Falling back to cloud provider: {provider}")
        try:
            if provider == "openai":
                payload = {
                    "model": OPENAI_MODEL,
                    "messages": messages,
                    "temperature": TEMPERATURE,
                }
                response = requests.post(
                    "https://api.openai.com/v1/chat/completions",
                    headers={
                        "Authorization": f"Bearer {OPENAI_API_KEY}",
                        "Content-Type": "application/json",
                    },
                    json=payload,
                    timeout=60,
                )
                response.raise_for_status()
                body = response.json()
                return body["choices"][0]["message"]["content"].strip()

            if provider == "gemini":
                prompt = "\n\n".join(
                    f"{message['role'].upper()}:\n{message['content']}" for message in messages
                )
                payload = {
                    "contents": [{"parts": [{"text": prompt}]}],
                    "generationConfig": {"temperature": TEMPERATURE},
                }
                response = requests.post(
                    (
                        "https://generativelanguage.googleapis.com/v1beta/models/"
                        f"{GEMINI_MODEL}:generateContent?key={GEMINI_API_KEY}"
                    ),
                    json=payload,
                    timeout=60,
                )
                response.raise_for_status()
                body = response.json()
                candidates = body.get("candidates", [])
                if candidates:
                    parts = candidates[0].get("content", {}).get("parts", [])
                    text = "".join(part.get("text", "") for part in parts).strip()
                    if text:
                        return text
        except Exception as exc:
            trace.append(f"[{agent_name}] Cloud fallback failed: {exc}")
        return None

def update_dimon(expert, latency):
    """
    DIMON PROTOCOL: Updates telemetry for real-time monitoring.
    """
    key = "coder" if expert == CODER_MODEL else "automator" if expert == AUTOMATOR_MODEL else "logician"
    DIMON_STATUS[key]["hits"] += 1
    # Simple moving average for latency
    old_avg = DIMON_STATUS[key]["avg_latency"]
    DIMON_STATUS[key]["avg_latency"] = (old_avg * 0.9) + (latency * 0.1)
    DIMON_STATUS["global"]["last_heartbeat"] = time.time()

    return DIMON_STATUS

def wrap_kqml(performative, content, sender="USER", receiver="AEGIS"):
    """
    KQML BUS ENVELOPE: Standardizes communication between MoE experts.

    msg_id = str(uuid.uuid4())[:8]

import hashlib
import json
from pathlib import Path

# SECURE PERMISSION MANIFEST
# In a real ceLoRA deployment, these hashes would be signed by the user.
ALLOWED_SHELL_COMMANDS = [
    "dir", "ls", "systeminfo", "git status", "git log", "python --version", "whoami"
]

ALLOWED_FILES = [
    "status.txt", "training_notes.txt", "algorithms.db", "research_report.md"
]

def generate_action_manifest(kind, payload):
    """
    Generates a cryptographic fingerprint for an intended action.
    """
    raw = f"{kind}:{payload}".encode("utf-8")
    return hashlib.sha256(raw).hexdigest()

def is_action_allowed(kind, payload, trace):
    """
    ZERO-TRUST VERIFIER: Checks if the AI's requested action matches
    pre-approved cryptographic manifests or safety patterns.
    """
    if kind == "shell":
        # Check against approved command list
        base_cmd = payload.split()[0].lower()
        if any(base_cmd in allowed for allowed in ALLOWED_SHELL_COMMANDS):
            return True
        trace.append(f"[ZERO-TRUST] BLOCKED: Shell command '{payload}' not in manifest.")
        return False
        
    if kind == "write":
        # Restrict writing to project data/txt files only
        filename = Path(payload).name
        if filename.endswith(".txt") or filename.endswith(".md") or filename.endswith(".json"):
            return True
        trace.append(f"[ZERO-TRUST] BLOCKED: Write access to '{filename}' denied.")
        return False

    return False

import sqlite3
import time
import json
from pathlib import Path
from typing import List, Tuple, Optional

ALGO_DB_PATH = Path(__file__).parent.parent / "data" / "algorithms.db"

# Algorithm promotion thresholds
PROMOTION_THRESHOLD = 10  # Uses before considering promotion
HIGH_WEIGHT_THRESHOLD = 5.0  # Weight threshold for "hot" algorithms
DECAY_RATE = 0.95  # Weight decay per hour for unused algorithms

def init_algo_db():
    ALGO_DB_PATH.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(ALGO_DB_PATH)
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS algorithms (
            name TEXT PRIMARY KEY,
            content TEXT,
            use_count INTEGER DEFAULT 0,
            total_tokens INTEGER DEFAULT 0,
            last_used REAL,
            weight REAL DEFAULT 1.0
        )
    """)
    conn.commit()
    conn.close()

def record_algorithm_use(name, content, tokens):
    """
    LAYER 1 INTERCEPTION: Logs algorithm usage and updates weights.
    """
    conn = sqlite3.connect(ALGO_DB_PATH)
    cursor = conn.cursor()
    now = time.time()
    
    # Upsert logic
    cursor.execute("""
        INSERT INTO algorithms (name, content, use_count, total_tokens, last_used)
        VALUES (?, ?, 1, ?, ?)
        ON CONFLICT(name) DO UPDATE SET
            use_count = use_count + 1,
            total_tokens = total_tokens + ?,
            last_used = ?,
            content = CASE WHEN ? != '' THEN ? ELSE content END
    """, (name, content, tokens, now, tokens, now, content, content))
    
    # Dynamic Weight Calculation with Promotion System
    # Formula: (use_count^1.2 * tokens) / (time_decay) + recency_boost
    cursor.execute("""
        UPDATE algorithms SET weight =
            (POWER(use_count, 1.2) * CAST(total_tokens AS REAL) / 1000.0) /
            (POWER(((? - last_used) / 3600.0 + 1.0), 0.8)) +
            CASE WHEN (? - last_used) < 3600 THEN 2.0 ELSE 0.0 END
    """, (now, now))
    
    conn.commit()
    conn.close()

def get_algorithm_prefetch(limit=5):
    """
    ALGO PREFETCH: Returns top weighted algorithms for context injection.
    """
    conn = sqlite3.connect(ALGO_DB_PATH)
    cursor = conn.cursor()
    cursor.execute("SELECT name, content FROM algorithms ORDER BY weight DESC LIMIT ?", (limit,))
    results = cursor.fetchall()
    conn.close()
    return results

def get_algo_stats():
    conn = sqlite3.connect(ALGO_DB_PATH)
    cursor = conn.cursor()
    cursor.execute("SELECT COUNT(*) FROM algorithms")
    count = cursor.fetchone()[0]
    cursor.execute("SELECT name, use_count, weight FROM algorithms ORDER BY weight DESC LIMIT 5")
    top = cursor.fetchall()
    conn.close()
    return {"total_algos": count, "top_algos": top}

def promote_hot_algorithms() -> List[Tuple[str, float]]:
    """
    Identify and promote frequently-used algorithms.
    Returns list of (name, weight) for algorithms above promotion threshold.
    """
    conn = sqlite3.connect(ALGO_DB_PATH)
    cursor = conn.cursor()
    cursor.execute("""
        SELECT name, weight, use_count
        FROM algorithms
        WHERE use_count >= ? AND weight >= ?
        ORDER BY weight DESC
    """, (PROMOTION_THRESHOLD, HIGH_WEIGHT_THRESHOLD))
    hot_algos = cursor.fetchall()
    conn.close()
    return [(name, weight) for name, weight, _ in hot_algos]

def decay_unused_algorithms():
    """
    Apply time-based decay to algorithm weights.
    Algorithms not used recently lose weight gradually.
    """
    conn = sqlite3.connect(ALGO_DB_PATH)
    cursor = conn.cursor()
    now = time.time()
    
    # Decay weights for algorithms not used in last hour
    cursor.execute("""
        UPDATE algorithms
        SET weight = weight * ?
        WHERE (? - last_used) > 3600
    """, (DECAY_RATE, now))
    
    affected = cursor.rowcount
    conn.commit()
    conn.close()
    return affected

def get_algorithm_recommendations(context: str, limit: int = 3) -> List[Tuple[str, str]]:
    """
    Get algorithm recommendations based on context and current weights.
    Returns list of (name, content) tuples.
    """
    conn = sqlite3.connect(ALGO_DB_PATH)
    cursor = conn.cursor()
    
    # Simple keyword matching + weight-based ranking
    context_lower = context.lower()
    cursor.execute("""
        SELECT name, content, weight
        FROM algorithms
        WHERE LOWER(name) LIKE ? OR LOWER(content) LIKE ?
        ORDER BY weight DESC
        LIMIT ?
    """, (f"%{context_lower}%", f"%{context_lower}%", limit))
    
    results = cursor.fetchall()
    conn.close()
    return [(name, content) for name, content, _ in results]
